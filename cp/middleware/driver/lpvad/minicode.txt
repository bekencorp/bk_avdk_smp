//VAD唤醒的DeepSleep流程
void sleep_ana_set_tmp(void){
       uint32_t val;
       
       val = addSYSTEM_Reg0x4a;
       val = val|(0x1 << 9);//spilatch
       write_ana_reg(0xa,val);  
       
       //val = addSYSTEM_Reg0x49;
       //val = val & ~(0x7 << 16) | (0x4 << 16) ; //vddaon ldo
       //write_ana_reg(0x9,val);      
       
       val = addSYSTEM_Reg0x4a;
       val = val & ~(0x7 << 20) | (0x2 << 20);  //vddcorel
       write_ana_reg(0xa,val);
       
       val = addSYSTEM_Reg0x4a;
       val = val & ~(0xf << 16) | (0xe << 16);  //vddcoreh
       write_ana_reg(0xa,val);
       
       val = addSYSTEM_Reg0x49;
       val = val & (~((0x1 << 21) | (0x1 << 10) | (0x1 << 2) | (0x1 << 1))) | 0x1; //low power mode enable 
       write_ana_reg(0x9,val);  
       
       val = addSYSTEM_Reg0x49;
       val = val | (0x1 << 1); //vddcore high power mode
       write_ana_reg(0x9,val);  
              
       val = addSYSTEM_Reg0x4a;
       val = val | (0x1 << 31) | (0x1 << 23); 
       write_ana_reg(0xa,val);
              
       val = addSYSTEM_Reg0x4a;
       val = val | (0x1 << 23); 
       write_ana_reg(0xa,val);
       
       val = addSYSTEM_Reg0x4c;
       val = val&(~(0x1 << 13));//close vdd_ana
       write_ana_reg(0xc,val);  

       val = addSYSTEM_Reg0x4a;
       val = val & ~(0x1 << 9);//close spilatch
       write_ana_reg(0xa,val);
       
       val = addSYSTEM_Reg0x45;
       val = (val & ~((0x1 << 5) | (0x1 << 3) | (0x1 << 2) | (0x1 << 1)));//close hf clk
       write_ana_reg(0x5,val);
       
       val = addSYSTEM_Reg0x45;
       val = (val & ~(0x1 << 6));//close hf clk
       write_ana_reg(0x5,val);  
}

void test_low_power_vad(uint8_t* p){
       uint32_t regData;

       //uint8_t sleep_delay_23_16 = p[0]; // 8 bit
       //uint8_t sleep_delay_15_8  = p[1]; // 8 bit
       //uint8_t sleep_delay_7_0   = p[2]; // 8 bit
       uint8_t mem_halt_pwd      = p[0];
       uint8_t config_vad        = p[1];
       
       //spilatch
       regData  = read_ana_reg(0xA);
       regData |=  (1<<9);
       write_ana_reg(0xA,regData);
       
       //Aon LDO
       regData  = read_ana_reg(0x9);
       regData &= ~(0x7<<16);
       regData |=  (0x6<<16);
       write_ana_reg(0x9,regData);
       
       //Core LDO
       regData  = read_ana_reg(0xA);
       regData &= ~((0x7<<19)|(0xF<<16));
       regData |=  ((0x6<<19)|(0x8<<16));
       write_ana_reg(0xA,regData);
       
       //CoreHS LDO
       regData  = read_ana_reg(0x10);
       regData &= ~((0x7<<15)|(0xF<<11));
       regData |=  ((0x6<<15)|(0x8<<11));
       write_ana_reg(0x10,regData);

       //spilatch
       regData  = read_ana_reg(0xA);
       regData &= ~(1<<9);
       write_ana_reg(0xA,regData);
       
       setf_SYSTEM_Reg0x4_cpu_int_mask;
       
       addSYSTEM_Reg0x10 |= 0xf;
       addAON_PMU_Reg0x42 |= 0x3;
              
       //Set Core and Flash clock to XTAL
       regData = addSYSTEM_Reg0x8;
       regData &= ~((0x7<<8)|(0x3<<6)|(0xF<<2)|(0x3<<0));
       regData |=  ((0x0<<8)|(0x0<<6)|(0x0<<2)|(0x0<<0));
       addSYSTEM_Reg0x8 = regData;

       //p_intn_func_handle[INT_ID_RTC] = &INT_HANDLER_SLEEP_AON_RTC;
       //p_intn_func_handle[INT_ID_VAD] = &INT_HANDLER_SLEEP_VAD;

       //rtc_wake_init2(5);  //rtc wake init    
       
       set_AON_PMU_Reg0x41_wakeup_ena(1<<(10-4));
       set_AON_PMU_Reg0x41_lpo_config(2);
       
       if(mem_halt_pwd & (1<<0)) set_AON_PMU_Reg0x41_halt_sram0(1);
       if(mem_halt_pwd & (1<<1)) set_AON_PMU_Reg0x41_halt_sram1(1);
       if(mem_halt_pwd & (1<<2)) set_AON_PMU_Reg0x41_halt_sram2(1);
       if(mem_halt_pwd & (1<<3)) set_AON_PMU_Reg0x41_halt_cache(1);
       set_AON_PMU_Reg0x41_mem_ret_en(1);
       
       sleep_ana_set_tmp();  //sleep ana set
       
       //sleep_halt_delay_set(0x1, sleep_delay_23_16, sleep_delay_15_8, sleep_delay_7_0);  //sleep_mode 0:lv_sleep 1:deep_sleep
       addAON_PMU_Reg0x43 |= 0x1 << 17;//clr wakeup_source
       //addAON_PMU_Reg0x40  =  0x4e000000 | (0x11 << 16) | (0x11 << 8) | (0x11 << 0);  //deep_sleep
       addSYSTEM_Reg0x11  |= 0x1 | (0x1 << 4);
       
       addAON_PMU_Reg0x0  = addAON_PMU_Reg0x0 & (~0x3) | 0x3;
       
       for(volatile gpio_i=0;gpio_i<72;gpio_i++){
              PMU_GPIO_CFG[gpio_i] = 0x0;
       }
       
       addAON_PMU_Reg0x0  |= (0x1 << 31);
       addAON_PMU_Reg0x25 = 0x424B55AA;
       addAON_PMU_Reg0x25 = 0xBDB4AA55;
       addAON_GPIO_Reg0x19 = (0x2 << 24);
       
       addAON_PMU_Reg0x41 &= ~(0x1 << 15);
       
       //VAD Config
       write_ana_reg(0x16,0x7CAA5241);
       write_ana_reg(0x17,0x40FE2A33);
       write_ana_reg(0x18,0x014064C0);
       
       regData  = read_ana_reg(0x2);
       regData &= ~(1<<31);
       write_ana_reg(0x2,regData);
       
       regData  = read_ana_reg(0x14);
       regData &= ~(1<<5);
       regData |=  (1<<5);
       write_ana_reg(0x14,regData);//MIC Bias
       regData  = read_ana_reg(0x1B);
       regData &= ~(1<<28);
       regData |=  (1<<28);
       write_ana_reg(0x1B,regData);//ADC en
       regData  = read_ana_reg(0x17);
       regData &= ~(1<<8);
       regData |=  (1<<8);
       write_ana_reg(0x17,regData);//VAD mode
       regData &= ~(1<<3);
       regData |=  (1<<3);
       write_ana_reg(0x17,regData);//VAD viniset
       waitUS(10000);
       regData &= ~(1<<2);
       regData |=  (1<<2);
       write_ana_reg(0x17,regData);//VAD rstn
       
              
       __ISB();
       __DSB();
       
       Delay(1000);
                     
       SetSleepMode(1);//0:CUP SLEEPING; 1: CPU SLEEPDEEP 
       
       //addAON_PMU_Reg0x41 &= ~(0x1 << 28);
       
       //Userprintf("addr %08x read %08x\r\n",0x44000008,reg_sel(0x44000008));
       //Userprintf("addr %08x read %08x\r\n",0x44000104,reg_sel(0x44000104));
              
       __ISB();
       __DSB();
       
       regData = addSYSTEM_Reg0x4a;
       regData = regData | (0x1 << 9) ;
       write_ana_reg(0xa,regData); 
       
       regData = addSYSTEM_Reg0x4c;
       //regData &= (~(0x1 << 13)) ; //vddana off when halt
       regData |=   (0x1 << 13)  ; //vddana on  when halt
       write_ana_reg(0xc,regData);
       
       regData = addSYSTEM_Reg0x49;
       regData = regData & ~(0x7 << 16) | ((0x1 & 0x7) << 16) ; //aon ldo
       write_ana_reg(0x9,regData);
       
       regData = addSYSTEM_Reg0x4a;
       regData = regData & ~(0x1 << 9) ;
       write_ana_reg(0xa,regData);
       
       __ISB();
       __DSB();
       
       __WFI();
