#include <stdio.h>
#include <unistd.h>
#include <fcntl.h>
#include "common.h"
#include "fs.h"

void main(void)
{
	int fd;
	int ch;

	/* initialize ctest core */
	ctest_core_init();

	fd = fs_open("inp.txt", O_RDWR);
	if (fd < 0) {
		printf("file open error\n");
		return 0;
	};

	printf("file open sucess\n\n");

	while (read(fd, &ch, 1) > 0)
		printf("%c", ch);

	printf("\nfile closed\n");
	close(fd);
}
